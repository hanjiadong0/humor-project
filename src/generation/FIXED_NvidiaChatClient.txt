class NvidiaChatClient(LLMClient):
    """
    Adapter for NVIDIA API to work with HumorPromptPipeline.
    Uses non-streaming for multi-step pipeline (cleaner for step1/step2).
    """
    def __init__(self, client, model: str, system_prompt: str):
        self.client = client
        self.model = model
        self.system_prompt = system_prompt

    def complete(self, prompt: str, *, temperature: float, max_tokens: int) -> str:
        """Execute a single LLM completion call."""
        resp = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": self.system_prompt},
                {"role": "user", "content": prompt},
            ],
            temperature=temperature,
            top_p=1.0,
            max_tokens=max_tokens,
            stream=False,
        )
        content = resp.choices[0].message.content
        # FIX: Handle None responses from API (but they're rare!)
        if content is None:
            print(f"WARNING: API returned None for prompt: {prompt[:50]}...")
            return ""
        return content
